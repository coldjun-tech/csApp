private enum FixtureViewState
{
    Unknown,
    Ready,
    Running,
    Alarm
}

// UI 스레드에서 호출
private void UpdateFixtureImage(
    PictureBox target, FixtureViewState state)
{
    Image nextImage;

    switch (state)
    {
        case FixtureViewState.Ready:
            nextImage = Properties.Resources.Blank_Badge_Green_5;
            break;

        case FixtureViewState.Running:
            nextImage = Properties.Resources.Blank_Disc_Blue_5;
            break;

        case FixtureViewState.Alarm:
            nextImage = Properties.Resources.Blank_Disc_Red_5;
            break;

        default:
            nextImage = Properties.Resources.Blank_Badge_Grey_5;
            break;
    }

    if (!ReferenceEquals(target.Image, nextImage))
        target.Image = nextImage;
}
