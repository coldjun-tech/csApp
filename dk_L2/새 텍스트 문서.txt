CREATE TABLE `tb_hmi_scrt` (
	`ScrtName` VARCHAR(50) NOT NULL DEFAULT '0' COLLATE 'utf8_general_ci',
	`SysName` VARCHAR(10) NOT NULL DEFAULT '0' COLLATE 'utf8_general_ci',
	`ScrtCatg` VARCHAR(30) NOT NULL DEFAULT '0' COLLATE 'utf8_general_ci',
	`WindowName` VARCHAR(50) NOT NULL DEFAULT '0' COLLATE 'utf8_general_ci',
	`ScrtText` TEXT NULL DEFAULT NULL COLLATE 'utf8_general_ci',
	INDEX `ScrtName_SysName_ScrtCatg_WindowName` (`ScrtName`, `SysName`, `ScrtCatg`, `WindowName`) USING BTREE
)
COLLATE='utf8_general_ci'
ENGINE=InnoDB
;
CREATE TABLE `tb_tag_data` (
	`TagID` INT(11) NOT NULL AUTO_INCREMENT,
	`TagName` VARCHAR(50) NOT NULL DEFAULT '0' COLLATE 'utf8_general_ci',
	`SysName` VARCHAR(50) NOT NULL DEFAULT '0' COLLATE 'utf8_general_ci',
	`GrpName` VARCHAR(50) NULL DEFAULT NULL COLLATE 'utf8_general_ci',
	`Comment` VARCHAR(100) NULL DEFAULT NULL COLLATE 'utf8_general_ci',
	PRIMARY KEY (`TagID`) USING BTREE,
	INDEX `TagName_System` (`TagName`, `SysName`) USING BTREE
)
COLLATE='utf8_general_ci'
ENGINE=InnoDB
AUTO_INCREMENT=15454
;
CREATE TABLE `tb_tag_ref` (
	`TagName` VARCHAR(50) NOT NULL DEFAULT '0' COLLATE 'utf8_general_ci',
	`SysName` VARCHAR(10) NOT NULL DEFAULT '0' COLLATE 'utf8_general_ci',
	`TagType` VARCHAR(30) NOT NULL DEFAULT '0' COLLATE 'utf8_general_ci',
	`TagUse` VARCHAR(50) NULL DEFAULT NULL COLLATE 'utf8_general_ci',
	`Position` VARCHAR(30) NULL DEFAULT NULL COLLATE 'utf8_general_ci',
	`WindowName` VARCHAR(50) NULL DEFAULT NULL COLLATE 'utf8_general_ci',
	`WhereScrt` VARCHAR(100) NULL DEFAULT NULL COLLATE 'utf8_general_ci',

COLLATE='utf8_general_ci'
ENGINE=InnoDB
;
